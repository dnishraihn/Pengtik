// ===================== normal/soal2_normal.cpp =====================
#include <iostream>
using namespace std;

// Dengan & : a dan b adalah ALIAS dari variabel asli
void tukar(int &a, int &b) {
    int temp = a;
    a = b;
    b = temp;
}

// Tanpa & : a dan b hanyalah SALINAN
void tukar_tanpa_ref(int a, int b) {
    int temp = a;
    a = b;
    b = temp;
}

int main() {
    int x = 5, y = 10;
    cout << "Awal            : x=" << x << " y=" << y << "\n";

    tukar_tanpa_ref(x, y);
    cout << "Tanpa &         : x=" << x << " y=" << y << "  (tidak berubah)\n";

    tukar(x, y);
    cout << "Dengan &        : x=" << x << " y=" << y << "  (tertukar)\n";
    return 0;
}

// ===================== modular/tukar.h =====================
#ifndef TUKAR_H
#define TUKAR_H

// Menukar nilai dua variabel (pass by reference)
void tukar(int &a, int &b);

// Versi SALAH untuk demonstrasi (pass by value)
void tukar_tanpa_ref(int a, int b);

#endif

// ===================== modular/tukar.cpp =====================
#include "tukar.h"

void tukar(int &a, int &b) {
    int temp = a;
    a = b;
    b = temp;
}

void tukar_tanpa_ref(int a, int b) {
    int temp = a;
    a = b;
    b = temp;
}

// ===================== modular/main.cpp =====================
#include <iostream>
#include "tukar.h"
using namespace std;

int main() {
    int x, y;
    cout << "Masukkan x dan y: ";
    cin >> x >> y;

    tukar_tanpa_ref(x, y);
    cout << "Tanpa &  : x=" << x << " y=" << y << "\n";

    tukar(x, y);
    cout << "Dengan & : x=" << x << " y=" << y << "\n";
    return 0;
}
